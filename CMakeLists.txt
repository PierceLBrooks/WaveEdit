    list(APPEND LANGUAGES "C")
    list(APPEND LANGUAGES "CXX")
    list(APPEND LANGUAGES "OBJC")
    list(APPEND LANGUAGES "OBJCXX")
    if ("${CMAKE_GENERATOR}" STREQUAL "Xcode")
        execute_process(COMMAND uname -m OUTPUT_VARIABLE UNAME_M ERROR_QUIET OUTPUT_STRIP_TRAILING_WHITESPACE)
        if (NOT "$ENV{SYSROOT}" STREQUAL "")
            set(SYSROOT "$ENV{SYSROOT}")
        else()
            execute_process(COMMAND xcodebuild -version -sdk macosx Path OUTPUT_VARIABLE SYSROOT ERROR_QUIET OUTPUT_STRIP_TRAILING_WHITESPACE)
        endif()
        if (NOT EXISTS "${SYSROOT}")
            message(FATAL_ERROR "SYSROOT does not exist!")
        endif()
        set(CMAKE_OSX_SYSROOT "${SYSROOT}")
        if ("${CMAKE_OSX_ARCHITECTURES}" STREQUAL "")
            if ("${UNAME_M}" MATCHES "arm")
                set(CMAKE_OSX_ARCHITECTURES "arm64")
            else()
                set(CMAKE_OSX_ARCHITECTURES "$(ARCHS_STANDARD)")
            endif()
        endif()
    endif()
else()
    list(APPEND LANGUAGES "C")
    list(APPEND LANGUAGES "CXX")
    if (MINGW)
        list(APPEND LANGUAGES "RC")
    endif()
endif()

    find_package(SampleRate REQUIRED)

else()
	set(DEPS )
	set(samplerate "libsamplerate-0.2.2")
	set(sndfile "libsndfile-1.2.2")
	set(sdl2 "SDL2-2.0.7")
	list(APPEND DEPS "samplerate")
	list(APPEND DEPS "sndfile")
	list(APPEND DEPS "sdl2")
    add_custom_target(libdep DEPENDS "dep")
    if (MINGW)
		list(APPEND LIBS "mingw32")
		add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/ext")
		if (TARGET Ogg::ogg)
			list(APPEND LIBS Ogg::ogg)
		endif()
		if (TARGET Vorbis::vorbis)
			list(APPEND LIBS Vorbis::vorbis)
		endif()
		if (TARGET Vorbis::vorbisenc)
			list(APPEND LIBS Vorbis::vorbisenc)
		endif()
        #add_custom_command(OUTPUT "dep" COMMAND "mingw32-make" ARGS "ARCH=win" "V=1" "VERBOSE=1" "MAKE=mingw32-make" "all" DEPENDS "${CMAKE_CURRENT_LIST_DIR}/dep/Makefile" COMMENT "Compile dependencies" WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/dep")
		if (NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/dep/lib")
			add_custom_target(dep DEPENDS "depsamplerate" "depsndfile" "depsdl2")
			if (TARGET Vorbis::vorbis)
				add_dependencies(dep Vorbis::vorbis)
			endif()
			if (TARGET Vorbis::vorbisenc)
				add_dependencies(dep Vorbis::vorbisenc)
			endif()
			add_custom_command(OUTPUT "depsamplerate" COMMAND "curl" ARGS "-L" "--insecure" "--verbose" "--output" "${samplerate}.tar.xz" "https://github.com/libsndfile/libsamplerate/releases/download/0.2.2/${samplerate}.tar.xz" DEPENDS "${CMAKE_CURRENT_LIST_DIR}/dep/Makefile" COMMENT "Fetch samplerate" WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/dep")
			add_custom_command(OUTPUT "depsndfile" COMMAND "curl" ARGS "-L" "--insecure" "--verbose" "--output" "${sndfile}.tar.xz" "https://github.com/libsndfile/libsndfile/releases/download/1.2.2/${sndfile}.tar.xz" DEPENDS "${CMAKE_CURRENT_LIST_DIR}/dep/Makefile" COMMENT "Fetch sndfile" WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/dep")
			add_custom_command(OUTPUT "depsdl2" COMMAND "curl" ARGS "-L" "--insecure" "--verbose" "--output" "${sdl2}.tar.gz" "https://www.libsdl.org/release/${sdl2}.tar.gz" DEPENDS "${CMAKE_CURRENT_LIST_DIR}/dep/Makefile" COMMENT "Fetch sdl2" WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/dep")
			foreach(D IN LISTS DEPS)
				if (NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/dep/${${D}}")
					if ("${D}" STREQUAL "sdl2")
						add_custom_command(OUTPUT "dep${D}" APPEND COMMAND "${CMAKE_COMMAND}" ARGS "-E" "tar" "xf" "${CMAKE_CURRENT_LIST_DIR}/dep/${${D}}.tar.gz" DEPENDS "${CMAKE_CURRENT_LIST_DIR}/dep/Makefile" COMMENT "Extract ${D}" WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/dep")
					else()
						add_custom_command(OUTPUT "dep${D}" APPEND COMMAND "${CMAKE_COMMAND}" ARGS "-E" "tar" "xf" "${CMAKE_CURRENT_LIST_DIR}/dep/${${D}}.tar.xz" DEPENDS "${CMAKE_CURRENT_LIST_DIR}/dep/Makefile" COMMENT "Extract ${D}" WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/dep")
					endif()
				endif()
				add_custom_command(OUTPUT "dep${D}" APPEND COMMAND "${CMAKE_COMMAND}" ARGS "-G" "${CMAKE_GENERATOR}" "-S" "${CMAKE_CURRENT_LIST_DIR}/dep/${${D}}" "-B" "${CMAKE_CURRENT_LIST_DIR}/dep/dep${${D}}" "-DDIRECTX=NO" "-DCMAKE_INSTALL_PREFIX=${CMAKE_CURRENT_LIST_DIR}/dep" "-DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}" "-DCMAKE_DEBUG_POSTFIX=${CMAKE_DEBUG_POSTFIX}" DEPENDS "${CMAKE_CURRENT_LIST_DIR}/dep/Makefile" COMMENT "Configure ${D}" WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/dep/${${D}}")
				add_custom_command(OUTPUT "dep${D}" APPEND COMMAND "${CMAKE_COMMAND}" ARGS "--build" "${CMAKE_CURRENT_LIST_DIR}/dep/dep${${D}}" "--target" "install" "--config" "${CMAKE_BUILD_TYPE}" "--verbose" DEPENDS "${CMAKE_CURRENT_LIST_DIR}/dep/Makefile" COMMENT "Compile ${D}" WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/dep/${${D}}")
			endforeach()
			set_source_files_properties("depsamplerate" PROPERTIES SYMBOLIC "true")
			set_source_files_properties("depsndfile" PROPERTIES SYMBOLIC "true")
			set_source_files_properties("depsdl2" PROPERTIES SYMBOLIC "true")
		endif()
    else()
		set(ARGS )
		#list(APPEND ARGS "ARCH=lin")
		if ("${CMAKE_BUILD_TYPE}" STREQUAL "Debug")
		    list(APPEND ARGS "V=1")
		    list(APPEND ARGS "VERBOSE=1")
		endif()
		foreach(D IN LISTS DEPS)
			list(APPEND ARGS "${D}=${${D}}")
		endforeach()
		list(APPEND ARGS "all")
        #add_custom_command(OUTPUT "dep" COMMAND "make" ARGS "V=1" "VERBOSE=1" "all" DEPENDS "${CMAKE_CURRENT_LIST_DIR}/dep/Makefile" COMMENT "Compile dependencies" WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/dep")
        add_custom_command(OUTPUT "dep" COMMAND "make" ARGS ${ARGS} DEPENDS "${CMAKE_CURRENT_LIST_DIR}/dep/Makefile" COMMENT "Compile dependencies" WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/dep")
    endif()
    set_source_files_properties("dep" PROPERTIES SYMBOLIC "true")
    add_library(libsamplerate STATIC IMPORTED)
    set_target_properties(libsamplerate PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/dep/lib/libsamplerate${CMAKE_DEBUG_POSTFIX}.a")
    add_library(libsndfile STATIC IMPORTED)
    set_target_properties(libsndfile PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/dep/lib/libsndfile${CMAKE_DEBUG_POSTFIX}.a")
    add_library(libsdl STATIC IMPORTED)
    set_target_properties(libsdl PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/dep/lib/libSDL2${CMAKE_DEBUG_POSTFIX}.a")
    add_library(libsdlmain STATIC IMPORTED)
    set_target_properties(libsdlmain PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/dep/lib/libSDL2main${CMAKE_DEBUG_POSTFIX}.a")
    list(APPEND LIBS libsamplerate)
    list(APPEND LIBS libsndfile)
    list(APPEND LIBS libsdl)
    if (MINGW)
        list(APPEND LIBS libsdlmain)
		list(APPEND LIBS "winmm")
		list(APPEND LIBS "imm32")
		list(APPEND LIBS "version")
		set_target_properties(libsdl PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/dep/lib/libSDL2-static${CMAKE_DEBUG_POSTFIX}.a")
    endif()
endif()
